until bq show --schema --format=prettyjson ${PROJECT_ID}:us_flow_logs.${TABLE} | grep -q src_instance; do echo "waiting for src_instance..."; sleep 30; done

# 5. The zone query
bq query --use_legacy_sql=false "
SELECT
  jsonPayload.src_instance.zone AS src_zone,
  jsonPayload.src_instance.vm_name AS src_vm,
  jsonPayload.dest_instance.zone AS dest_zone,
  jsonPayload.dest_instance.vm_name AS dest_vm
FROM \`${PROJECT_ID}.us_flow_logs.compute_googleapis_com_vpc_flows_*\`
